29-- 3.查找sell表中平均订购册数在5本以上的用户号和平均订购册数


30select 用户号,avg(订购册数) from sell group by 用户号 having avg(订购册数)>5;

32-- 查找sell表中已经发货平均册数在5本以上的用户号和平均册数


33select 用户号,avg(订购册数) from sell where 是否发货 = '已发货' group by 用户号 having avg(订购册数) > 5;

35-- 查找sell表中会员订单数在2笔以上且每笔订购册数都在5本以上的会员


36select 用户号,count(*) from sell where 订购册数 >5 group by 用户号 having count(*) >=2;
