25	语法: select 需要查找字段名(英文逗号隔开) from 所有字段所在的表名(多个表，英文逗号隔开) where【条件是多个表内字段相等】

29select 书名,订购册数,订购时间 from book,sell


30where book.`图书编号` = sell.`图书编号`;


31


32-- 查找购买mysql数据库且订购数量大于5的图书信息


33


34select book.* from book,sell 


35WHERE book.`图书编号` = sell.`图书编号`


36and book.`书名` = 'MYSQL数据库' AND sell.`订购册数` > 5;


37


38-- 查询订购数量大于5本图书的会员姓名和订购册数


39


40select 姓名 as '会员姓名',订购册数 from members,sell


41where members.`用户号` = sell.`用户号` and sell.`订购册数` > 5;


42


43-- 查询购买"网页程序设计" 且订购数量大于5本图书和会员姓名和订购册数


44


45select 图书，姓名 as 会员姓名，订购册数 from book,members,sell


46where book.`图书编号` = sell.`图书编号` and sell.`用户号` = members.`用户号`


47and book.`书名` = '网页程序设计' and sell.`订购册数` > 5;
