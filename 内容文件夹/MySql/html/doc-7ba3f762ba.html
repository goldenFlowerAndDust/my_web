3/*			一、新建表时添加约束


4									1. 添加字段级约束：在字段名和类型


5									2.主键 (primary key)  ,不重复不为null，一个表只有一个


6									3.check(条件) 检查约束


7									4.unique 替代主建，不重复可为null，一个表可多个


8									5.default 默认值


9				二、表级约束：


10									1.在字段最下面[CONSTRAINT 约束名] 约束类型(字段名)


11				三、外键约束：


12									1.外表/主表 ，子表、默认restrict拒绝修改


13									2.cascade 自动更新


14									


15*/


16


17create table student(


18sno char(7) primary key,


19sname varchar(10) not NULL,


20sex char(1) check(sex='男' or sex = '女'),


21seat int unique,


22	age int default 18


23);


24


25create table student1(


26sno char(7),


27sname varchar(10) not NULL,


28sex char(1),


29seat int unique,


30age int default 18,


31PRIMARY key(sno),


32check(sex='男' or sex='女')


33);


34


35create table student2(


36sno char(7),


37sname varchar(10) not NULL,


38sex char(1),


39seat int unique,


40age int default 18,


41PRIMARY key(sno),


42check(sex='男' or sex='女'),


43FOREIGN key(sno) REFERENCES student(sno)


44);

46create table student3(


47sno char(7),


48sname varchar(10) not NULL,


49sex char(1),


50seat int unique,


51age int default 18,


52PRIMARY key(sno),


53check(sex='男' or sex='女'),


54FOREIGN key(sno) REFERENCES student1(sno) on delete CASCADE on UPDATE CASCADE
