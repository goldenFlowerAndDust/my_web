1.请编写程序求出10以内的质数

	<!DOCTYPE html>
	<html lang="en">
	<head>
		<meta charset="UTF-8">
		<title>求10以内质数</title>
	</head>
	<body>
	<div id="prime"></div>
	<script type="text/javascript">
		// 获取元素
		let prime = document.getElementById("prime")
		// 新建输出，用于接受结果
		let Primes = []

		// 质数：大于1，只有1和自身两个因数：说白了就是余数是自身或1

		for (let i = 2; i <= 10; i++) {
			// 外层循环：除数——最后结果值
			let isPrime = true // 初始化布尔值，用于后续判断

			for (let j = 2; j <= Math.sqrt(i); j++) { // 求平方根，保证是奇数，偶数没有质数
				// 内层循环：被除数
				if (i % j === 0) {
					// 当余数为0，则不是质数
					isPrime = false // 将布尔值替换为false
					break; // 遇到第一个，退出循环，降低性能
				}

			}
			if (isPrime) Primes.push(i) // 做判断，当布尔值为true时，将除数放入数组
		}
		prime.innerHTML = `10以内的质数是：${Primes.join()}`
	</script>
	</body>
	</html> 


2.	将一个数组arr=[2, 0, 6, 1, 77, 0, 52, 0, 25, 7]中所有数值为0的元素删除。
	<script type=”text/javascript”>
		// 删除数值为0的元素，可以使用筛选(filter)——不改变原数组
		// 也可以使用，splice()——改变原数组
		方法一：
			arr.filter((item) => {
		return item !==0 // 只需要筛选出0即可
	})
		方法二：
			for(let i = 0; i < arr.length; i++) {
				// i是索引
			If (arr[i] === 0){
				arr.splice(i,1) // 删除当前索引————对应的就是0
		} 
	}
	</script>
	

3. 编写完整HTML+原生JS代码，实现功能： 
•1.页面有一个文本输入框、【保存临时数据】按钮
•2.点击按钮，将输入框内容保存到 sessionStorage 中，键名自定义为 tempData
•3.保存成功后弹出提示“临时数据保存成功”
•4.刷新页面后，读取sessionStorage数据并展示在页面div中

	<!DOCTYPE html>
	<html lang="en">
	<head>
		<meta charset="UTF-8">
		<title>临时保存数据</title>
	</head>
	<body>
	<label>
		<input type="text" id="text">
		<input type="button" id="btn" value="保存临时数据">
	</label>
	<div>
		<button type="button" id="go0">刷新</button>
		<div id="content"></div>
	</div>
	<script>
		// 获取用户输入值
		let text = document.getElementById("text") // 因为是表单，读取使用value
		// 获取按钮，并添加时间监听
		document.getElementById("btn").addEventListener('click', () => {
			let Text = text.value // 实时获取用户输入的值
			// 使用本地存储(sessionStorage)
			//增/改：setItem()，查：getItem() ,删：removeItem() 清除：clear()
			sessionStorage.setItem('tempData', `${Text}`) // 键默认：tempData 值为用户输入的
			alert("临时数据保存成功") // 弹出提示框
			text.value = '' // 单击按钮清空文本输入框
		})
		// 第二部分：刷新显示存储"
		// 获取数据显示节点
		let content = document.getElementById("content")
		// 获取刷新按钮
		document.getElementById("go0").addEventListener("click", () => {
			// 获取 getItem
			let value = sessionStorage.getItem("tempData")
			// 键获取到的值，显示出来
			content.innerHTML = `存储的临时数据是：${value}`
		})
	</script>
	</body>
	</html>
